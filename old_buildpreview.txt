function buildPreviewHTML(){
  const inv=curInv()
  const gallery=S.gallery||[]
  if(!inv)return'<html><body style="font-family:sans-serif;padding:40px;color:#999;text-align:center"><p>Keine Einladung ausgewählt</p></body></html>'

  const hf=inv.font_heading||'Great Vibes'
  const bf=inv.font_body||'Lato'
  const tplId=inv.font_preset||'classic'
  const tpl=TEMPLATES[tplId]||TEMPLATES.classic
  const gold=inv.theme_primary_color||tpl.gold
  const bg=inv.theme_bg_color||tpl.bg
  const hasImg=!!inv.hero_image_url
  const n1=inv.host_name_1||'Name 1', n2=inv.host_name_2||'Name 2'

  // Calendar
  let calHTML=''
  let countdownTarget=''
  if(inv.event_date){
    const d=new Date(inv.event_date+'T12:00:00')
    const yr=d.getFullYear(),mo=d.getMonth(),dy=d.getDate()
    const months=['Januar','Februar','März','April','Mai','Juni','Juli','August','September','Oktober','November','Dezember']
    const wdays=['Mo','Di','Mi','Do','Fr','Sa','So']
    const first=new Date(yr,mo,1).getDay(),offset=first===0?6:first-1
    const total=new Date(yr,mo+1,0).getDate()
    let cells=''
    for(let i=0;i<offset;i++)cells+=`<div></div>`
    for(let dd=1;dd<=total;dd++){
      const active=dd===dy
      cells+=`<div style="aspect-ratio:1;display:flex;align-items:center;justify-content:center;font-size:11px;font-weight:300;color:${active?'white':'rgba(250,248,243,.6)'};border-radius:50%;background:${active?gold:'transparent'};${active?`box-shadow:0 0 14px ${gold}80`:''};">${dd}</div>`
    }
    // Countdown target ISO string for JS
    const hm=(inv.ceremony_time||'14:00:00').slice(0,5)
    countdownTarget=`${inv.event_date}T${hm}:00`
    calHTML=`<div style="background:${tpl.darkBg||'#262019'};padding:56px 20px;text-align:center;color:white">
      <div style="font-family:'${hf}',serif;font-size:clamp(26px,5vw,42px);font-style:italic;color:${gold};margin-bottom:28px;letter-spacing:.06em">${months[mo]} ${yr}</div>
      <div style="max-width:280px;margin:0 auto">
        <div style="display:grid;grid-template-columns:repeat(7,1fr);gap:2px;margin-bottom:6px">${wdays.map(w=>`<div style="font-size:8.5px;text-transform:uppercase;color:rgba(250,248,243,.35);text-align:center;padding:2px 0">${w}</div>`).join('')}</div>
        <div style="display:grid;grid-template-columns:repeat(7,1fr);gap:2px">${cells}</div>
      </div>
      <div style="display:flex;gap:10px;justify-content:center;margin-top:36px;flex-wrap:wrap">
        ${[['cd-d','Tage'],['cd-h','Std.'],['cd-m','Min.'],['cd-s','Sek.']].map(([id,l])=>`<div style="background:rgba(250,248,243,.07);border:1px solid rgba(250,248,243,.14);border-radius:12px;padding:14px 16px;text-align:center;min-width:62px"><div id="${id}" style="font-family:'${hf}',serif;font-size:32px;font-weight:400;color:#FAF8F3;line-height:1">…</div><div style="font-size:9px;letter-spacing:.12em;text-transform:uppercase;opacity:.45;margin-top:4px;color:white">${l}</div></div>`).join('')}
      </div>
    </div>`
  }

  // Gallery
  const pics=gallery.filter(g=>g.url)
  const galHTML=pics.length>0?`<div style="display:grid;grid-template-columns:repeat(2,1fr);gap:3px">${pics.slice(0,4).map(g=>`<div style="aspect-ratio:4/3;overflow:hidden;background:#E5D8C3"><img src="${g.url}" style="width:100%;height:100%;object-fit:cover;object-position:${g.position_x||50}% ${g.position_y||50}%;transform:scale(${g.scale||1});transform-origin:center"></div>`).join('')}</div>`:''

  // Countdown script — runs inside iframe
  const cdScript = countdownTarget ? `<script>
(function(){
  var target=new Date('${countdownTarget}').getTime();
  function pad(n){return String(n).padStart(2,'0')}
  function tick(){
    var diff=Math.max(0,target-Date.now());
    var d=Math.floor(diff/86400000),h=Math.floor(diff%86400000/3600000),
        m=Math.floor(diff%3600000/60000),s=Math.floor(diff%60000/1000);
    ['cd-d','cd-h','cd-m','cd-s'].forEach(function(id,i){
      var e=document.getElementById(id);
      if(e)e.textContent=pad([d,h,m,s][i]);
    });
  }
  tick();setInterval(tick,1000);
})();
<\/script>` : ''

  return`<!DOCTYPE html><html><head><meta charset="UTF-8"><meta name="viewport" content="width=device-width,initial-scale=1">
<link rel="stylesheet" href="https://fonts.googleapis.com/css2?family=${hf.replace(/ /g,'+')}:ital,wght@0,300;0,400;1,400&family=${bf.replace(/ /g,'+')}:wght@300;400&display=swap">
<style>*{box-sizing:border-box;margin:0;padding:0}body{font-family:'${bf}',sans-serif;background:${bg};color:#262019;overflow-x:hidden;-webkit-font-smoothing:antialiased;line-height:1.6}:root{--gold:${gold}}</style>
</head><body>
<!-- HERO -->
<div style="position:relative;min-height:100vh;display:flex;align-items:flex-end;overflow:hidden;background:${hasImg?'#5C4838':'linear-gradient(150deg,'+tpl.heroGrad+')'}">
  ${hasImg?`<img src="${inv.hero_image_url}" style="position:absolute;inset:0;width:100%;height:100%;object-fit:cover;object-position:${inv.hero_pos_x||50}% ${inv.hero_pos_y||50}%;transform:scale(${inv.hero_scale||1});transform-origin:center"><div style="position:absolute;inset:0;background:linear-gradient(to bottom,rgba(38,32,25,.1),rgba(38,32,25,.7))"></div>`:''}
  <div style="position:relative;z-index:1;text-align:center;padding:52px 24px 72px;width:100%;color:white">
    <div style="font-size:9.5px;letter-spacing:.3em;text-transform:uppercase;opacity:.6;margin-bottom:18px">— Einladung zur Hochzeit —</div>
    <div style="font-family:'${hf}',serif;font-size:clamp(48px,9vw,96px);font-weight:300;line-height:1.05;letter-spacing:.02em">${n1}<br><span style="font-style:italic;opacity:.72;font-size:.65em">&amp;</span><br>${n2}</div>
    ${inv.tagline?`<div style="font-size:14px;font-weight:300;opacity:.7;margin-top:10px;letter-spacing:.08em">${inv.tagline}</div>`:''}
  </div>
</div>
${inv.quote_enabled&&inv.quote_text?`<div style="background:${tpl.quoteBg||'linear-gradient(150deg,#F3EDE0,#FAF8F3)'};padding:60px 32px;text-align:center;position:relative;overflow:hidden"><div style="position:absolute;top:-30px;left:14px;font-family:'${hf}',serif;font-size:180px;font-weight:300;color:#E5D8C3;opacity:.5;line-height:1">\u201C</div>${inv.quote_heading?`<div style="font-size:10px;letter-spacing:.22em;text-transform:uppercase;color:${gold};margin-bottom:16px">${inv.quote_heading}</div>`:''}<div style="font-family:'${hf}',serif;font-size:clamp(20px,4vw,32px);font-weight:300;font-style:italic;color:#3F3828;line-height:1.55;margin-bottom:14px;position:relative">${inv.quote_text}</div>${inv.quote_source?`<div style="font-size:12.5px;color:#AE8E6C">— ${inv.quote_source}</div>`:''}</div>`:''}
${calHTML}
${(inv.ceremony_venue_name||inv.ceremony_address)?`<div style="background:${tpl.ceremonyBg||'#FEFCF9'};padding:60px 24px;text-align:center"><div style="width:66px;height:66px;border-radius:50%;background:linear-gradient(135deg,#F3EDE0,#E5D8C3);display:flex;align-items:center;justify-content:center;font-size:26px;margin:0 auto 20px;box-shadow:0 8px 24px rgba(38,32,25,.08)">${{church:'⛪',mosque:'🕌',registry:'🏛️',garden:'🌿',beach:'🌊',custom:'🏡'}[inv.ceremony_venue_type]||'📍'}</div><div style="font-family:'${hf}',serif;font-size:clamp(26px,4vw,40px);font-weight:400;color:#262019;margin-bottom:20px">${inv.ceremony_title||'Zeremonie'}</div>${inv.ceremony_venue_name?`<div style="font-size:17px;font-weight:500;color:#262019;margin-bottom:6px">${inv.ceremony_venue_name}</div>`:''} ${inv.ceremony_time_str?`<div style="display:inline-flex;align-items:center;gap:7px;background:#F3EDE0;border-radius:100px;padding:10px 22px;font-size:15px;font-weight:500;margin:14px 0">🕐 ${inv.ceremony_time_str} Uhr</div>`:''} ${inv.ceremony_address?`<div style="font-size:14px;color:#AE8E6C;font-weight:300;margin-top:8px">${inv.ceremony_address}</div>`:''} ${inv.ceremony_map_url?`<div style="margin-top:18px"><a href="${inv.ceremony_map_url}" style="display:inline-flex;align-items:center;gap:7px;background:${gold};color:white;border-radius:100px;padding:12px 26px;font-size:13.5px;text-decoration:none">📍 Route anzeigen</a></div>`:''}</div>`:''}
${inv.welcome_note?`<div style="background:${bg};padding:56px 24px;text-align:center"><div style="font-family:'${hf}',serif;font-size:clamp(17px,2.5vw,24px);font-weight:300;font-style:italic;color:#3F3828;line-height:1.7;max-width:500px;margin:0 auto">${inv.welcome_note.replace(/\n/g,'<br>')}</div></div>`:''}
${(inv.reception_enabled&&(inv.timeline_json||[]).length>0)?`<div style="background:${tpl.timelineBg||'#F3EDE0'};padding:56px 24px"><div style="font-family:'${hf}',serif;font-size:clamp(26px,4vw,38px);font-weight:400;color:#262019;text-align:center;margin-bottom:40px">${inv.reception_title||'Ablauf'}</div><div style="max-width:440px;margin:0 auto;padding-left:44px;position:relative"><div style="position:absolute;left:21px;top:6px;bottom:6px;width:1px;background:linear-gradient(to bottom,transparent,#E5D8C3 10%,#E5D8C3 90%,transparent)"></div>${(inv.timeline_json||[]).map(item=>`<div style="position:relative;margin-bottom:28px"><div style="position:absolute;left:-44px;width:40px;height:40px;border-radius:50%;background:white;border:1.5px solid #E5D8C3;display:flex;align-items:center;justify-content:center;font-size:15px;box-shadow:0 2px 7px rgba(38,32,25,.06)">${item.icon||'🎉'}</div><div style="font-size:10.5px;letter-spacing:.13em;text-transform:uppercase;color:${gold};margin-bottom:3px">${item.time||''}</div><div style="font-size:15px;font-weight:500;color:#262019;margin-bottom:2px">${item.title||''}</div><div style="font-size:13px;color:#AE8E6C;font-weight:300">${item.venue||''}</div></div>`).join('')}</div></div>`:''}
${galHTML}
${inv.rsvp_enabled?`<div style="background:${tpl.rsvpBg||'#262019'};padding:60px 24px;text-align:center"><div style="font-size:10px;letter-spacing:.22em;text-transform:uppercase;color:${gold};margin-bottom:13px">✦ Rückmeldung ✦</div><div style="font-family:'${hf}',serif;font-size:clamp(28px,5vw,46px);font-weight:300;color:${tpl.rsvpTitle||'#FAF8F3'};margin-bottom:6px">Bist du dabei?</div><div style="font-size:14px;color:rgba(250,248,243,.45);font-weight:300;margin-bottom:26px">Wir freuen uns auf deine Antwort</div><div style="max-width:380px;margin:0 auto"><div style="background:rgba(250,248,243,.07);border:1.5px solid rgba(250,248,243,.14);border-radius:9px;padding:12px 14px;font-size:13.5px;color:rgba(250,248,243,.3);margin-bottom:9px;text-align:left">Dein vollständiger Name</div><div style="background:linear-gradient(135deg,${gold},${gold}cc);color:white;border-radius:9px;padding:14px;font-size:14px;text-align:center;letter-spacing:.04em">Rückmeldung bestätigen →</div></div></div>`:''}
<div style="background:#262019;padding:24px;text-align:center"><span style="font-family:'Cormorant Garamond',serif;font-size:18px;color:rgba(250,248,243,.3);letter-spacing:.06em">Festlich<span style="color:${gold}">.</span></span></div>
${cdScript}
</body></html>`
}
